-- Chạy một lần trong Supabase SQL Editor để bật Hỏi đáp Trang chủ Đạo Vũ. create table if not exists public.public_questions ( id bigint generated by default as identity primary key, name text not null, phone text not null, question text not null, answer text, status text not null default 'pending' check (status in ('pending','answered')), created_at timestamptz not null default now(), answered_at timestamptz ); alter table public.public_questions enable row level security; drop policy if exists "Public can submit questions" on public.public_questions; create policy "Public can submit questions" on public.public_questions for insert to anon with check ( length(trim(name)) between 1 and 120 and length(trim(phone)) between 6 and 30 and length(trim(question)) between 1 and 3000 and status = 'pending' and answer is null ); drop policy if exists "Public can read answered questions" on public.public_questions; create policy "Public can read answered questions" on public.public_questions for select to anon using (status = 'answered'); drop policy if exists "Admins can read all questions" on public.public_questions; create policy "Admins can read all questions" on public.public_questions for select to authenticated using ( exists ( select 1 from public.profiles p where lower(p.email) = lower(auth.jwt() ->> 'email') and p.role = 'admin' ) ); drop policy if exists "Admins can answer questions" on public.public_questions; create policy "Admins can answer questions" on public.public_questions for update to authenticated using ( exists ( select 1 from public.profiles p where lower(p.email) = lower(auth.jwt() ->> 'email') and p.role = 'admin' ) ) with check ( exists ( select 1 from public.profiles p where lower(p.email) = lower(auth.jwt() ->> 'email') and p.role = 'admin' ) ); grant insert on public.public_questions to anon; grant select on public.public_questions to anon, authenticated; grant update on public.public_questions to authenticated; grant usage, select on sequence public.public_questions_id_seq to anon;
